def HPC.Fold.S3 : Prop :=
∀ (n : ℕ) (ρ σ : Fin n → ℤ) (i : Fin n) (H L : ℤ),
  IsLip ρ →
    IsCrookedIvF ρ σ →
      (∀ (t : Fin n), L ≤ ρ t) →
        (∀ (t : Fin n), ρ t ≤ H) →
          L + 3 ≤ H → (∃ t, ρ t = L) → (∃ t, ρ t = H) → ∃ len c, FoldsOnto (armG ρ i) len c ∧ FoldsOnto (armT ρ i) len c
s3_general : S3
@theorem3 : ∀ {X : Type u_1} [inst : MetricSpace X] {n k : ℕ},
  0 < n →
    ∀ {E : HPC.Chain X n} {D : HPC.Chain X k},
      E.IsCrookedIn' D →
        ∀ (i : Fin n),
          ∃ q β',
            ∃ (hβ' : HPC.Chain.IsBlockMap β'),
              ↑(β' i) = 0 ∧ ∀ (u : Fin q), ∃ j, (E.ofBlocks β' hβ').link u ⊆ D.unionRange j 3
